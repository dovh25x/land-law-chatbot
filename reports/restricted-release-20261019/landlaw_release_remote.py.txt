import json, sys, hashlib
from pathlib import Path
import httpx
from dotenv import dotenv_values

ROOT=Path('/home/dovh25x/projects/land-law-chatbot')
OUT=ROOT/'reports/restricted-release-20261019'
OUT.mkdir(exist_ok=True)
env=dotenv_values(ROOT/'.env')
sid=env['RENDER_SERVICE_ID']
with httpx.Client(base_url='https://api.render.com/v1', timeout=45,
        headers={'Authorization':'Bearer '+env['RENDER_API_KEY']}) as c:
    r=c.get('/services/'+sid);r.raise_for_status()
    service=r.json();d=service['serviceDetails']
    assert d['plan']=='free' and d['runtime']=='docker'
    assert service['repo'].removesuffix('.git')=='https://github.com/dovh25x/land-law-chatbot'
    r=c.get(f'/services/{sid}/env-vars?limit=100');r.raise_for_status()
    current={x['envVar']['key']:x['envVar']['value'] for x in r.json()}
    deployments=c.get(f'/services/{sid}/deploys?limit=5');deployments.raise_for_status()
    report={'plan':d['plan'],'runtime':d['runtime'],'url':d['url'],
            'auto_deploy':service.get('autoDeployTrigger',service.get('autoDeploy')),
            'generation_enabled_before':current.get('GENERATION_ENABLED','true'),
            'configured_groq_key_matches_local':current.get('GROQ_API_KEY')==env.get('GROQ_API_KEY'),
            'deployments':[{'id':x['deploy']['id'],'status':x['deploy']['status'],
                'commit':x['deploy'].get('commit',{}).get('id')} for x in deployments.json()]}
    if sys.argv[1]=='restrict':
        assert report['auto_deploy'] in ('off','no')
        r=c.put(f'/services/{sid}/env-vars/GENERATION_ENABLED',json={'value':'false'});r.raise_for_status()
        r=c.put(f'/services/{sid}/env-vars/GROQ_API_KEY',json={'value':env['GROQ_API_KEY']});r.raise_for_status()
        r=c.get(f'/services/{sid}/env-vars?limit=100');r.raise_for_status()
        after={x['envVar']['key']:x['envVar']['value'] for x in r.json()}
        assert after['GENERATION_ENABLED']=='false'
        assert after['GROQ_API_KEY']==env['GROQ_API_KEY']
        report['generation_enabled_configured']=False
        report['groq_key_synchronized']=True
        report['guard']='Existing Settings.generation_enabled: status unavailable and authenticated chat HTTP 503, including historical dates.'
    (OUT/('render-'+sys.argv[1]+'.json')).write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps(report))
