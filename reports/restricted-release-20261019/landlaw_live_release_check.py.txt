import hashlib, json, sys
from pathlib import Path
import httpx
from dotenv import dotenv_values
from landlaw.config import Settings
from landlaw.evaluation import configuration_fingerprint
from landlaw.security import utcnow

ROOT=Path('/home/dovh25x/projects/land-law-chatbot');OUT=ROOT/'reports/restricted-release-20261019'
env=dotenv_values(ROOT/'.env');base=env['VITE_API_BASE_URL']
with httpx.Client(timeout=55,follow_redirects=True) as c:
    home=c.get(base);home.raise_for_status()
    def get(path):
        r=c.get(base+path);r.raise_for_status();return r.json()
    ready=get('/readyz');status=get('/api/v1/status');evaluation=get('/api/v1/evaluations/latest');docs=get('/api/v1/documents')['documents']
    assert ready['database'] and ready['storage']=='postgresql'
    assert status['generation_available'] is False and status['reason']=='disabled'
    assert evaluation['active_configuration_sha256']==configuration_fingerprint(Settings())
    assert ready['corpus_release']==status['corpus_release']==evaluation['active_release']==sys.argv[1]
    assert all(d.get('project') in (None,'ring4') for d in docs)
    direct=c.get('https://land-law-chatbot.onrender.com/api/v1/status')
    assert direct.status_code==403 and direct.json()['detail']['code']=='untrusted_ingress'
    cf=c.get(f"https://api.cloudflare.com/client/v4/accounts/{env['CLOUDFLARE_ACCOUNT_ID']}/pages/projects/{env['CLOUDFLARE_PAGES_PROJECT']}/deployments",headers={'Authorization':'Bearer '+env['CLOUDFLARE_API_TOKEN']})
    cf.raise_for_status();deployments=cf.json();assert deployments['success']
    latest=next(d for d in deployments['result'] if d['environment']=='production' and d['latest_stage']['status']=='success')
    assert latest['deployment_trigger']['metadata']['commit_hash']=='6ca8f12f0d770f2076de20f0d070460894bd0bc8'
    report={'checked_at':utcnow().isoformat(),'frontend':base,'frontend_http':home.status_code,
        'readyz':ready,'generation_status':status,'documents':len(docs),
        'scope_verified':True,'runtime_configuration_sha256':evaluation['active_configuration_sha256'],
        'evaluation_matches_active_corpus':evaluation['matches_active_release'],
        'evaluation_matches_active_configuration':evaluation['matches_active_configuration'],
        'unsigned_backend_request_http':direct.status_code,'generation_model_live_tested':False,
        'cloudflare':{'id':latest['id'],'environment':latest['environment'],'url':latest['url'],
            'commit':latest['deployment_trigger']['metadata']['commit_hash'],'status':latest['latest_stage']['status']}}
    (OUT/sys.argv[2]).write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps(report,ensure_ascii=False))
